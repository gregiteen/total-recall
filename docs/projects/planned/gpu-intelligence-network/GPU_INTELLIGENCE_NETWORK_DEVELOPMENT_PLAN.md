## 2. GPU Market Intelligence Engine

### A. SkyPilot Catalog Adapter (Primary Data Source)

SkyPilot's `skypilot-catalog` (Apache 2.0) provides the richest, most reliable pricing data:

```javascript
// src/core/gpu-intelligence/providers/skypilot.mjs
import fs from 'fs/promises';
import path from 'path';
import { logger } from '../logger.mjs';

const CATALOG_URL = 'https://raw.githubusercontent.com/skypilot-org/skypilot-catalog/master/catalogs';

export async function fetchSkypilotPricing() {
  const providers = ['aws', 'gcp', 'azure', 'runpod', 'lambda', 'vast', 'coreweave'];
  const results = [];
  
  for (const provider of providers) {
    try {
      const url = `${CATALOG_URL}/${provider}/vms.csv`;
      const response = await fetch(url);
      if (!response.ok) continue;
      
      const csv = await response.text();
      const offers = parseSkypilotCSV(csv, provider);
      results.push(...offers);
    } catch (err) {
      logger.warn('gpu-intel', `SkyPilot fetch failed for ${provider}: ${err.message}`);
    }
  }
  
  return results;
}

function parseSkypilotCSV(csv, provider) {
  // SkyPilot CSV columns: InstanceType, AcceleratorName, AcceleratorCount, vCPUs, MemoryGiB, Price, SpotPrice, Region
  const lines = csv.trim().split('\n');
  const headers = lines[0].split(',');
  
  return lines.slice(1).map(line => {
    const cols = line.split(',');
    return {
      provider,
      instance_type: cols[headers.indexOf('InstanceType')],
      gpu_name: cols[headers.indexOf('AcceleratorName')],
      gpu_count: parseInt(cols[headers.indexOf('AcceleratorCount')] || '1'),
      vcpus: parseInt(cols[headers.indexOf('vCPUs')] || '0'),
      memory_gib: parseFloat(cols[headers.indexOf('MemoryGiB')] || '0'),
      price_hourly: parseFloat(cols[headers.indexOf('Price')] || '0'),
      spot_price_hourly: parseFloat(cols[headers.indexOf('SpotPrice')] || '0'),
      region: cols[headers.indexOf('Region')]
    };
  }).filter(o => o.gpu_name && o.price_hourly > 0);
}
```

## 3. Distributed Research Protocol

### A. Multi-Client Claim/Submit

Every Total Recall daemon instance participates in distributed GPU intelligence gathering:

```
POST /api/research
{
  "topic": "gpu-pricing-vast-ai-h100",
  "priority": "medium",
  "notes": "Scrape current Vast.ai H100 offers via API. Report avg/min/max hourly rates and availability %.",
  "x_research_type": "gpu-market-intelligence",
  "x_provider": "vast-ai",
  "x_gpu_model": "H100"
}
```

**Claim protocol:**
1. Daemon calls `GET /api/research?status=pending&query=gpu-pricing` to find available tasks
2. Claims task via `PATCH /api/research/:id` → `{ status: "in_progress" }`
3. Executes provider scrape
4. POSTs results as SSSS memory nodes via `POST /api/memory`
5. Marks task as `done` with result summary

**De-duplication:**
- Nodes use deterministic slugs (`gpu-offer-{provider}-{gpu}-{region}`)
- Subsequent writes to same slug perform an update (not a duplicate)
- `x_verified_at` timestamp tracks freshness
- Nodes with `decay.half_life_days: 3` auto-expire if not refreshed

### B. Contributor Tracking

Each daemon identifies itself with a client fingerprint:

```javascript
const CLIENT_ID = crypto.createHash('sha256')
  .update(os.hostname() + os.platform() + os.arch())
  .digest('hex')
  .slice(0, 12);
```

Contributor counts are tracked in a derived index (`memory-derived/gpu-contributors.jsonl`) for the leaderboard.

```javascript
export function generateRecommendation(answers, marketData, modelCatalog) {
  // 1. Filter models by hard constraints (context, multimodal, privacy)
  let candidates = modelCatalog.filter(m => {
    if (answers.context === '1M+' && (m.x_context_window || 0) < 1_000_000) return false;
    if (answers.multimodal === 'text+image' && !m.supports_vision) return false;
    return true;
  });
  
  // 2. Filter by budget
  const budgetMap = { 'free': 0, '$50/mo': 50, '$200/mo': 200, '$500/mo': 500, '$2000+/mo': 10000 };
  const maxBudget = budgetMap[answers.budget] || 0;
  
  // 3. Score remaining candidates by use-case fit
  candidates = candidates.map(model => {
    let score = 0;
    if (answers.use_case === 'coding' && model.supports_code) score += 30;
    if (answers.use_case === 'agents' && model.supports_tools) score += 30;
    score += (model.x_benchmark_scores?.humaneval || 0) * 100;
    
    // Find cheapest hardware that runs this model
    const cheapestOffer = findCheapestOffer(model, marketData, maxBudget);
    if (!cheapestOffer) return null;
    
    return { model, offer: cheapestOffer, score };
  }).filter(Boolean);
  
  // 4. Sort by score and return top recommendation + alternatives
  candidates.sort((a, b) => b.score - a.score);
  
  return {
    primary: candidates[0] || null,
    alternatives: candidates.slice(1, 4)
  };
}
```

### B. JIT Capacity Broker

```javascript
// src/core/gpu-intelligence/broker.mjs
export async function provisionBestGpu(requirements) {
  const { minVram, maxBudget, preferredRegion, reliability } = requirements;
  
  // 1. Query live market data from vault
  const offers = loadGpuOfferNodes(VAULT_DIR)
    .filter(o => o.x_vram_gb >= minVram)
    .filter(o => o.x_monthly_rate <= maxBudget)
    .sort((a, b) => {
      // Weight: 40% price, 30% reliability, 20% availability, 10% latency
      const scoreA = (1/a.x_hourly_rate * 0.4) + ((a.x_reliability_score || 0.5) * 0.3);
      const scoreB = (1/b.x_hourly_rate * 0.4) + ((b.x_reliability_score || 0.5) * 0.3);
      return scoreB - scoreA;
    });
  
  if (offers.length === 0) throw new Error('No GPU offers match requirements');
  
  // 2. Attempt provision with failover
  for (const offer of offers.slice(0, 3)) {
    try {
      const adapter = getAdapter(offer.x_gpu_provider);
      const instance = await adapter.provision(offer);
      return instance;
    } catch (err) {
      logger.warn('broker', `Provision failed on ${offer.x_gpu_provider}: ${err.message}`);
      continue;
    }
  }
  
  throw new Error('All provision attempts failed');
}
```
